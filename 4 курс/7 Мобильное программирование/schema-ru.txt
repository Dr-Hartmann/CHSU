Project "ТрекерТренировок" {
  database_type: 'SQLite'
  Note: 'Схема базы данных для приложения отслеживания тренировок.'
}

// --- Основные Пользователь и Настройки ---

Table Пользователи {
  id integer [pk, increment]
  имя varchar
  создан_в datetime
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']
}

Table Настройки {
  id_пользователя integer [pk]
  язык varchar
  тема varchar
  длительность_таймера_отдыха integer
  недельные_лимиты json [note: 'JSON с лимитами для групп мышц']
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']
  
  Indexes {
    id_пользователя
  }
}

// --- Логирование Тренировок ---

Table Тренировки {
  id uuid [pk]
  id_пользователя integer
  дата datetime
  создан_в datetime
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_пользователя
  }
}

Table ГруппыУпражнений {
  id uuid [pk]
  id_тренировки uuid
  индекс_порядка integer
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_тренировки
  }
}

Table ЗаписиУпражнений {
  id uuid [pk]
  id_группы_упражнений uuid
  id_упражнения varchar // Ссылка на Упражнения.id
  порядок_в_группе integer
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_группы_упражнений
  }
}

Table ЗаписиПодходов {
  id uuid [pk]
  id_записи_упражнения uuid
  метрики json [note: 'например, {повторения, вес} или {дистанция, время}']
  является_разминочным boolean
  id_родительского_подхода uuid [note: 'Может быть NULL. Для дроп-сетов']
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_записи_упражнения
  }
}

// --- Тело и Шаблоны ---

Table ИзмеренияТела {
  id uuid [pk]
  id_пользователя integer
  дата datetime
  вес_кг float [null]
  процент_жира_в_теле float [null]
  грудь_см float [null]
  талия_см float [null]
  бедра_см float [null]
  левая_рука_см float [null]
  правая_рука_см float [null]
  правое_бедро_см float [null]
  левое_бедро_см float [null]
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_пользователя
  }
}

Table ШаблоныТренировок {
  id uuid [pk]
  id_пользователя integer
  имя varchar
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_пользователя
  }
}

Table ШаблонныеГруппыУпражнений {
  id uuid [pk]
  id_шаблона uuid
  индекс_порядка integer
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_шаблона
  }
}

Table ШаблонныеУпражнения {
  id uuid [pk]
  id_шаблонной_группы_упражнений uuid
  id_упражнения varchar // Ссылка на Упражнения.id
  порядок_в_группе integer
  обновлен_в integer [note: 'Unix timestamp']
  удален bool [default: 0, note: '0 или 1']

  Indexes {
    id_шаблонной_группы_упражнений
  }
}

// --- Статические Определения и Достижения (меньшая сложность синхронизации) ---

Table Упражнения {
  id varchar [pk, note: "например, 'жим_лежа'"]
  ключ_имени varchar [note: 'Ссылка на ключ I18N']
  тип_записи varchar [note: 'вес, кардио, время']
}

Table МышечныеГруппыУпражнений {
  id_упражнения varchar [pk] // Ссылка на Упражнения.id
  id_мышечной_группы varchar [pk] // Ссылка на МышечныеГруппы.id
  является_основной boolean [note: 'Определяет основную целевую мышцу']
}

Table МышечныеГруппы {
  id varchar [pk, note: "например, 'Грудь'"]
  ключ_имени varchar [note: 'Ссылка на ключ I18N']
}

Table Достижения {
  id varchar [pk, note: "например, 'первая_тренировка'"]
  ключ_имени varchar [note: 'Ключ I18N']
  ключ_описания varchar [note: 'Ключ I18N']
}

Table ДостиженияПользователей {
  id_пользователя integer [pk]
  id_достижения varchar [pk]
  разблокировано_в datetime [note: 'Только добавление, простая синхронизация']
  
  Indexes {
    id_пользователя
  }
}
